// Basic Usage - Optimized Build
import bzip2 from './dist/optimized-bzip2.js';

// Initialize with automatic optimization detection
await bzip2.initialize();

// Compress data (auto block size selection)
const data = new TextEncoder().encode('Hello World!');
const compressed = bzip2.compress(data);

// Decompress
const decompressed = bzip2.decompress(compressed);
const result = new TextDecoder().decode(decompressed);

console.log('Original:', data.length, 'bytes');
console.log('Compressed:', compressed.length, 'bytes');
console.log('Ratio:', (data.length / compressed.length).toFixed(2), 'x');

// Advanced Performance Tuning
// Fast compression for small data
const fastCompressed = bzip2.compress(data, 1);

// Maximum compression for archival
const maxCompressed = bzip2.compress(data, 9);

// Streaming compression for large files  
const chunk1 = bzip2.compressStream(data1, BZ_RUN);
const chunk2 = bzip2.compressStream(data2, BZ_RUN);
const final = bzip2.compressStream(new Uint8Array(0), BZ_FINISH);

// Performance monitoring
const stats = bzip2.getPerformanceStats();
console.log('Avg compression speed:', stats.averageCompressionSpeed, 'KB/s');
console.log('SIMD operations used:', stats.simdOperationsUsed);
